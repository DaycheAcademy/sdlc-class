CMAPPS Dataset Download:

1 - Set-Location "D:\Dayche\SDLC\class_files" 
2 - New-Item -ItemType Directory -Force -Path ".\data\raw\cmapps"
3 - Invoke-WebRequest -Uri "https://phm-datasets.s3.amazonaws.com/NASA/6.+Turbofan+Engine+Degradation+Simulation+Data+Set.zip" -OutFile ".\data\raw\cmapps\cmapps.zip"
4 - Get-Item ".\data\raw\cmapps\cmapps.zip"
5 - Expand-Archive -Path ".\data\raw\cmapps\cmapps.zip" -DestinationPath ".\data\raw\cmapps" -Force
6 - Get-ChildItem ".\data\raw\cmapps" -Recurse
